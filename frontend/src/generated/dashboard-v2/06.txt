      <button className="contest-main" onClick={onOpen} aria-label={`Open details for ${contest.title}`}>
        <div className="score-block"><strong>{contest.status === "BLOCKED" ? "—" : contest.score}</strong><span>score</span></div>
        <div className="contest-copy">
          <div className="contest-meta"><span className={`status-pill status-${contest.status.toLowerCase()}`}>{contest.status}</span><span>{contest.locality}</span><span>{contest.confidence} confidence</span></div>
          <h2>{contest.title}</h2>
          <p>{contest.prize}</p>
        </div>
        <div className="card-metric"><span>Estimated field</span><b>{crowdLabel(contest)}</b><small>{chanceRange(contest)}</small></div>
        <div className="card-metric deadline"><span>Deadline</span><b>{formatDeadline(contest.deadline)}</b><small>{deadlineLabel(contest.deadline)}</small></div>
        <Icon name="arrow" size={20} />
      </button>
      <div className="contest-actions">
        <span>{contest.friction}</span><span>{contest.winners} winner{contest.winners === 1 ? "" : "s"}</span><span>{contest.competition} competition</span>
        <div />
        {!["ENTERED", "BLOCKED", "SKIPPED"].includes(contest.status) && (
          <>
            <button onClick={() => onStatus(contest.id, "SKIPPED")}><Icon name="skip" size={14} />Skip</button>
            <button className="entered-action" onClick={() => onStatus(contest.id, "ENTERED")}><Icon name="check" size={14} />Entered</button>
          </>
        )}
      </div>
    </article>
  );
}

function DetailDialog({ contest, onClose, onStatus }: { contest: Contest; onClose: () => void; onStatus: (id: number, status: ContestStatus) => void }) {

  const evidence = parseEvidence(contest.evidenceUrls);

            <div className="probability-card">
              <span>Estimated probability</span><strong>{crowdLabel(contest)}</strong>
              <p>{chanceRange(contest)} · likely {formatChance(contest.chanceLikelyPpm)} · roughly {contest.entrantsLow.toLocaleString("en-GB")}–{contest.entrantsHigh.toLocaleString("en-GB")} entrants</p>

          </div>

          <section className="research-note"><span className="status-dot is-connected" /><div><b>Research note</b><p>{contest.verification}</p></div></section>

          {evidence.length > 0 && <div className="source-links">{evidence.slice(0, 5).map((url, index) => <a href={url} target="_blank" rel="noreferrer" key={url}>Source {index + 1}<Icon name="external" size={14} /></a>)}</div>}
        </div>

        <footer className="detail-footer">
          <a href={contest.url} target="_blank" rel="noreferrer">Open contest<Icon name="external" size={17} /></a>
          {!["ENTERED", "BLOCKED"].includes(contest.status) && <button onClick={() => onStatus(contest.id, "ENTERED")}><Icon name="check" size={17} />Mark entered</button>}
        </footer>
      </section>
    </div>
  );
}
