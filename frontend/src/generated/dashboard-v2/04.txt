      announce({ kind: "error", message });
      setToast(message);
    } finally {
      window.clearInterval(timer);
      window.setTimeout(() => {
        setScanning(false);
        setScanProgress(0);
      }, 450);
    }
  }

  async function runHunt(event: FormEvent) {
    event.preventDefault();
    const wanted = brief.trim();
    if (!wanted || hunting || !requireBackend()) return;
    setHunting(true);
    try {
      const response = await fetch(`${railway.url.replace(/\/$/, "")}/v1/hunt`, {
        method: "POST",
        headers: {
          "content-type": "application/json",
          authorization: `Bearer ${railway.secret}`,
        },
        body: JSON.stringify({
          brief: wanted,
          known_urls: contests.map((contest) => contest.url),
          known_titles: contests.map((contest) => contest.title),
          limit: 16,
          profile: PROFILE,
        }),
      });
      const payload = (await response.json()) as {
        interpretation?: string;
        adjacent_queries_used?: boolean;
        result?: DiscoveryResponse;
        detail?: string;
      };
      if (!response.ok || !payload.result) {
        throw new Error(payload.detail ?? "Hunt failed");
      }
      const found = Array.isArray(payload.result.discoveries)
        ? payload.result.discoveries
        : [];
      const imported = found.map((item, index) => mapDiscovery(item, -Date.now() - index));
      const existing = new Set(contests.map((contest) => normalizeContestUrl(contest.url)));
      const novel = imported.filter((contest) => !existing.has(normalizeContestUrl(contest.url)));
      if (novel.length) setContests((current) => [...novel, ...current]);
      setFilter("ALL");
      setSelectedId(null);
      setBrief("");
      const widened = payload.adjacent_queries_used ? " The search widened to adjacent prize categories." : "";
      setToast(
        novel.length
          ? `${novel.length} new matches for “${payload.interpretation ?? wanted}”.${widened}`
          : `Nothing new for “${wanted}”. Try a broader description.${widened}`,
      );
    } catch (error) {
      setToast(error instanceof Error ? error.message : "Hunt failed.");
    } finally {
      setHunting(false);
    }
  }

  function exportBackup() {
    const backup: Backup = {
      version: 2,
      exported_at: new Date().toISOString(),
      contests,
      railway,
      discovery_round: discoveryRound,
    };
    const blob = new Blob([JSON.stringify(backup, null, 2)], { type: "application/json" });
    const href = URL.createObjectURL(blob);
    const anchor = document.createElement("a");
    anchor.href = href;
    anchor.download = `eye-of-loki-${new Date().toISOString().slice(0, 10)}.json`;
    document.body.append(anchor);
    anchor.click();
    anchor.remove();
    URL.revokeObjectURL(href);
    setToast("Backup exported. It includes the private Railway secret.");
  }

  async function importBackup(file: File) {
    try {
      const parsed = JSON.parse(await file.text()) as Partial<Backup>;
      const imported = sanitizeContestList(parsed.contests, FALLBACK_CONTESTS);
      if (!imported.length) throw new Error("invalid");
      const config = {
        url: typeof parsed.railway?.url === "string" ? parsed.railway.url : "",
        secret: typeof parsed.railway?.secret === "string" ? parsed.railway.secret : "",
      };
      setContests(imported);
      setRailway(config);
      setDiscoveryRound(
        Number.isInteger(parsed.discovery_round) ? Number(parsed.discovery_round) : 0,
      );
      setSettingsOpen(false);
      setSelectedId(null);
      setConnection(config.url && config.secret ? "unknown" : "offline");
      setToast("Backup imported.");
    } catch {
      setToast("That file is not a valid Eye of Loki backup.");
    }
  }
