  function updateStatus(id: number, status: ContestStatus) {
    setContests((current) =>
      current.map((contest) =>
        contest.id === id
          ? {
              ...contest,
              status,
              enteredAt: status === "ENTERED" ? new Date().toISOString() : contest.enteredAt,
              verdict:
                status === "ENTERED"
                  ? "Entered"
                  : status === "SKIPPED"
                    ? "Skipped"
                    : contest.verdict,
            }
          : contest,
      ),
    );
    setToast(status === "ENTERED" ? "Added to your active portfolio." : "Removed from the active queue.");
  }

  async function testConnection(config = railway, quiet = false) {
    if (!config.url.trim()) {
      setConnection("offline");
      if (!quiet) setToast("Add the Railway service URL first.");
      return false;
    }
    setTesting(true);
    try {
      const response = await fetch(`${config.url.replace(/\/$/, "")}/health`);
      const payload = (await response.json()) as {
        ok?: boolean;
        gemini_configured?: boolean;
        search_configured?: boolean;
        auth_configured?: boolean;
        capabilities?: string[];
      };
      const healthy = Boolean(
        response.ok &&
          payload.ok &&
          payload.gemini_configured &&
          payload.search_configured &&
          payload.auth_configured &&
          payload.capabilities?.includes("discovery"),
      );
      setConnection(healthy ? "connected" : "offline");
      if (!quiet) {
        setToast(
          healthy
            ? payload.capabilities?.includes("adaptive-discovery")
              ? "Website-form discovery is online."
              : "Backend is online, but still running the old discovery engine."
            : "Backend answered, but one or more capabilities or secrets are missing.",
        );
      }
      return healthy;
    } catch {
      setConnection("offline");
      if (!quiet) setToast("Could not reach the Railway service.");
      return false;
    } finally {
      setTesting(false);
    }
  }

  function requireBackend() {
    if (railway.url.trim() && railway.secret.trim()) return true;
    setSettingsOpen(true);
    setToast("Connect the private Railway backend first.");
    return false;
  }

  async function verifyPendingLeads(current: Contest[]) {
    const pending = current
      .filter(
        (contest) =>
          contest.analysisMethod === "unscored" || needsEnglishTranslation(contest),
      )
      .slice(0, 8);
    if (!pending.length) return current;
    try {
      const response = await fetch(`${railway.url.replace(/\/$/, "")}/v1/recon`, {
        method: "POST",
        headers: {
          "content-type": "application/json",
          authorization: `Bearer ${railway.secret}`,
        },
        body: JSON.stringify({
          contests: pending.map((contest) => ({
            id: contest.id,
            slug: contest.slug,
            title: contest.title,
            organizer: contest.organizer,
            prize: contest.prize,
            url: contest.url,
            deadline: contest.deadline,
            winners: contest.winners,
            locality: contest.locality,
            eligibility: contest.eligibility,
            entry_method: contest.entryMethod,
            known_registration_required: contest.registrationRequired,
          })),
          profile: PROFILE,
        }),
      });
      const payload = (await response.json()) as { analyses?: ReconAnalysis[] };
      if (!response.ok || !payload.analyses) return current;
      const byId = new Map(payload.analyses.map((analysis) => [analysis.contest_id, analysis]));
      return current.map((contest) => {
        const analysis = byId.get(contest.id);
        return analysis ? mapAnalysis(contest, analysis) : contest;
      });
    } catch {
      return current;
    }
  }
